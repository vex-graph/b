cmake_minimum_required(VERSION 3.24)
# Standalone CLion metadata only; b bootstraps/builds itself, not through CMake.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(b_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(GLOB sources CONFIGURE_DEPENDS "*.c" "*.h" "adapters/*.c" "adapters/*.h")
add_library(b_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(b_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(b_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(b_ide PRIVATE -mcpu=apple-m1)
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${resource}/include/stddef.h")
        target_compile_options(b_ide PRIVATE -isystem "${resource}/include")
    endif()
endif()
message(STATUS "IDE metadata only: excluded object target; ./b is the build entry. No linking, downloads or runner.")
